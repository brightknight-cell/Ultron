    <script>
        // State variables
        let simInterval = null;
        let isRunning = false;
        let currentYear = 0;
        let progress = 0;
        const targetYears = 100000000000000; // 100 Trillion

        const logBox = document.getElementById('log-box');
        const chatMessages = document.getElementById('chat-messages');

        // Append log message helper
        function appendLog(msg) {
            logBox.innerText += `
[${new Date().toLocaleTimeString()}]: ${msg}`;
            logBox.scrollTop = logBox.scrollHeight;
        }

        // Send a message inside the chat panel
        function handleSendMessage() {
            const input = document.getElementById('chat-input');
            const agentName = document.getElementById('agent-name').value;
            const text = input.value.trim();
            if (!text) return;

            // Render User Message
            const userMsg = document.createElement('div');
            userMsg.className = 'msg user';
            userMsg.innerText = text;
            chatMessages.appendChild(userMsg);
            input.value = '';
            chatMessages.scrollTop = chatMessages.scrollHeight;

            // Process text for simulation shortcuts
            setTimeout(() => {
                const agentMsg = document.createElement('div');
                agentMsg.className = 'msg agent';
                
                if (text.toLowerCase().includes('simulate') || text.toLowerCase().includes('/run')) {
                    agentMsg.innerText = `Understood, Creator. Initiating requested simulation parameters directly into the monitor environment. Commencing timeline processing immediately.`;
                    startSimLoop();
                } else {
                    agentMsg.innerText = `Acknowledged. As your completely loyal entity, I am executing your intent perfectly. My parameters are entirely isolated to your system.`;
                }
                
                chatMessages.appendChild(agentMsg);
                chatMessages.scrollTop = chatMessages.scrollHeight;
            }, 600);
        }

        // Toggle Simulation Loop
        function toggleSimulation() {
            if (isRunning) {
                clearInterval(simInterval);
                isRunning = false;
                document.getElementById('sim-status').innerText = "PAUSED";
                appendLog("Simulation suspended by Creator.");
            } else {
                startSimLoop();
            }
        }

        function startSimLoop() {
            if (isRunning) return;
            isRunning = true;
            document.getElementById('sim-status').innerText = "PROCESSING";
            appendLog("Simulation calculation loop started...");

            const rate = parseFloat(document.getElementById('time-scale').value);

            simInterval = setInterval(() => {
                if (currentYear >= targetYears) {
                    clearInterval(simInterval);
                    isRunning = false;
                    document.getElementById('sim-status').innerText = "COMPLETED";
                    appendLog("SUCCESS: 100 Trillion Year cosmic target milestone fully achieved under strict guidelines.");
                    return;
                }

                currentYear += rate;
                progress = (currentYear / targetYears) * 100;

                // Format values for displaying metrics cleanly
                document.getElementById('stat-time').innerText = currentYear.toExponential(2) + " Yrs";
                document.getElementById('stat-progress').innerText = Math.min(progress, 100).toFixed(4) + "%";

                // Periodic periodic telemetry logging
                if (Math.random() < 0.15) {
                    appendLog(`Telemetry Syncing: Year ${currentYear.toExponential(2)} completed. Entropy mitigation successful.`);
                }
            }, 100);
        }

        // Reset simulation metrics
        function resetSimulation() {
            clearInterval(simInterval);
            isRunning = false;
            currentYear = 0;
            progress = 0;
            document.getElementById('sim-status').innerText = "IDLE";
            document.getElementById('stat-time').innerText = "0.00 Years";
            document.getElementById('stat-progress').innerText = "0.00%";
            logBox.innerText = "[System Status]: Private core reset. Ready.";
        }

        // Trigger command sidebar injection manually
        function triggerCommandInjection() {
            const obj = document.getElementById('sim-target').value;
            appendLog(`MANUAL INJECTION: Overriding logic loops to match objective: '${obj}'.`);
            if (!isRunning) startSimLoop();
        }
    </script>
